rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    match /agendamentos/{document} {
      allow create: if true;
      allow read: if true;
      allow update, delete: if request.auth != null;
    }

    match /servicos/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /bloqueiosAgenda/{document} {
      allow read: if true;
      allow create, update, delete: if request.auth != null;
    }

    match /logsSistema/{document} {
      allow create: if request.resource.data.origem == 'Agendamento Online'
                    && request.resource.data.mensagem is string
                    && request.resource.data.mensagem.size() <= 1200;
      allow read, update, delete: if request.auth != null;
    }

    match /prospectCallbacks/{document} {
      allow create: if request.resource.data.origem == 'Agendamento Online'
                    && request.resource.data.telefone is string
                    && request.resource.data.telefoneNormalizado is string
                    && request.resource.data.status == 'Pendente';
      allow update: if request.auth != null
                    || (
                      resource.data.origem == 'Agendamento Online'
                      && request.resource.data.telefone == resource.data.telefone
                      && request.resource.data.telefoneNormalizado == resource.data.telefoneNormalizado
                      && request.resource.data.origem == resource.data.origem
                      && request.resource.data.status in ['Pendente', 'Convertido']
                    );
      allow read, delete: if request.auth != null;
    }

    // Resumo público mínimo do Clube: apenas progresso e status do prêmio.
    // O formulário pode ler somente um documento específico e reservar um
    // benefício disponível. Listagem e manutenção completa ficam no Admin.
    match /clubePetlyneResumo/{document} {
      allow get: if true;
      allow list, create, delete: if request.auth != null;

      allow update: if request.auth != null || (
        request.resource.data.telefoneNormalizado == resource.data.telefoneNormalizado
        && request.resource.data.banhosTotal == resource.data.banhosTotal
        && request.resource.data.progresso == resource.data.progresso
        && request.resource.data.cicloAtual == resource.data.cicloAtual
        && (
          (
            resource.data.hidratacaoStatus == 'Disponivel'
            && request.resource.data.hidratacaoStatus == 'Reservado'
            && request.resource.data.hidratacaoCiclo == resource.data.hidratacaoCiclo
            && request.resource.data.banhoGratisStatus == resource.data.banhoGratisStatus
            && request.resource.data.banhoGratisCiclo == resource.data.banhoGratisCiclo
            && request.resource.data.hidratacaoProtocolo is string
            && request.resource.data.hidratacaoPet is string
          )
          ||
          (
            resource.data.banhoGratisStatus == 'Disponivel'
            && request.resource.data.banhoGratisStatus == 'Reservado'
            && request.resource.data.banhoGratisCiclo == resource.data.banhoGratisCiclo
            && request.resource.data.hidratacaoStatus == resource.data.hidratacaoStatus
            && request.resource.data.hidratacaoCiclo == resource.data.hidratacaoCiclo
            && request.resource.data.banhoGratisProtocolo is string
            && request.resource.data.banhoGratisPet is string
          )
        )
      );
    }

    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
